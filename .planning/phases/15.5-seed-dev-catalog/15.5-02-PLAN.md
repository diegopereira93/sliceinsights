```python
# --- Market-offer ingestion ---
from app.db.database import sync_engine
from app.db.ingestor import ingest_rows
from app.models.store import Store
from sqlmodel import Session, select

STORE_NAME = "Just Paddles"
DOMAIN = "justpaddles.com.br"

async def fetch_justpaddles_products():
    """Fetch product listings from Just Paddles category page."""
    from playwright.async_api import async_playwright

    products = []
    async with async_playwright() as p:
        browser = await p.chromium.launch(headless=True)
        page = await browser.new_page()

        category_urls = [
            f"https://{DOMAIN}/collections/pickleball",
            f"https://{DOMAIN}/collections/paddles",
        ]
        for url in category_urls:
            try:
                resp = await page.goto(url, timeout=15000)
                if not resp or resp.status >= 400:
                    continue
                await page.wait_for_load_state("networkidle", timeout=10000)
                await page.wait_for_timeout(2000)

                cards = await page.query_selector_all('[data-product-id], .product-item, .product-card')
                for card in cards:
                    title_el = await card.query_selector('a.product-title, h3 a, .title a, h3')
                    price_el = await card.query_selector('.price, .product-price')
                    link_el = await card.query_selector('a[href*="/products/"]')

                    title = await title_el.inner_text() if title_el else ""
                    price_str = await price_el.inner_text() if price_el else "0"
                    link = await link_el.get_attribute('href') if link_el else ""

                    if not title or "/products/" not in link:
                        continue

                    brand, model = _parse_title(title)
                    if not brand or not model:
                        continue

                    price = _parse_price(price_str)
                    if price <= 0:
                        continue

                    products.append({
                        "brand_name": brand,
                        "model_name": model,
                        "price_brl": str(price),
                        "product_url": f"https://{DOMAIN}{link}",
                        "image_url": "",
                    })
            except Exception as e:
                print(f"⚠️  Error fetching {url}: {e}")
                continue

        await browser.close()

    print(f"🛒 Found {len(products)} products from Just Paddles")
    return products

def _parse_title(title: str):
    """Split 'Brand Model' or 'Model' from title text."""
    title = title.strip()
    known_brands = [
        "Selkirk", "Joola", "Onix", "Engage", "Franklin", "Paddletek",
        "ProXR", "Gamma", "Head", "Babolat", "Nike", "Adidas",
        "ProSpin", "Dbear", "Kleren", "Voltrics", "Rally",
    ]
    title_lower = title.lower()
    for brand in known_brands:
        if title_lower.startswith(brand.lower()):
            model = title[len(brand):].strip("- ").strip()
            return brand, model
    return None, title

def _parse_price(text: str) -> float:
    """Extract BRL price from text like 'R$ 1.299,90' or '1299.90'."""
    import re
    text = text.replace("R$", "").replace("$", "").strip()
    # Remove thousands separator, keep decimal
    text = re.sub(r"(\d)\.(\d{3})", r"\1", text)
    text = text.replace(",", ".")
    try:
        return float(text)
    except ValueError:
        return 0.0

async def run_market_offer_ingestion():
    """Main entry point for market-offer ingestion."""
    print(f"🏓 Scraping {STORE_NAME}...")

    products = await fetch_justpaddles_products()
    if not products:
        print("⚠️  No products found, skipping ingestion")
        return

    with Session(sync_engine) as session:
        store = session.exec(
            select(Store).where(Store.name == STORE_NAME)
        ).one()

        stats = ingest_rows(products, store_id=store.id, session=session)
        session.commit()

        print(f"\n{'=' * 60}")
        print(f"✅ JUST PADDLES INGESTION COMPLETE")
        print(f"   🟢 Created:  {stats['created']}")
        print(f"   🔄 Updated: {stats['updated']}")
        print(f"   ⏭️  Skipped: {stats['skipped']}")

async def main_ingestion():
    await run_market_offer_ingestion()
```
