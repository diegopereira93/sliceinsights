### 2. Verificar DB após cada scraping
```bash
docker exec picklematch_api_v3 python -c "
from app.db.database import sync_engine
from app.models.market_offer import MarketOffer
from app.models.paddle import PaddleMaster
from sqlmodel import Session, select, func

with Session(sync_engine) as s:
    total = s.exec(select(func.count(MarketOffer.id))).one()
    brands = s.exec(select(func.count(func.distinct(MarketOffer.paddle_id)))).one()
    stores = s.exec(select(func.count(func.distinct(MarketOffer.store_id)))).one()
    print(f'MarketOffers: {total}')
    print(f'Unique paddles: {brands}')
    print(f'Stores with offers: {stores}')
"
```
