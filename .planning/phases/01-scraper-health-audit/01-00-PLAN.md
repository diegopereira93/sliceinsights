def init_test_db(seed=False, dry_run=False):
    """
    Reset test database to clean state.

    Args:
        seed: If True, add minimal test data
        dry_run: If True, print SQL without executing

    Returns:
        dict with status: "success", "error", or "dry-run"
    """
    try:
        from sqlmodel import create_engine, SQLModel, Session, select
        from app.models import PaddleMaster  # Adjust import to match project structure

        # Use test database URL
        db_url = os.environ.get(
            "TEST_DATABASE_URL",
            "postgresql://postgres:postgres@postgres_v3:5432/picklematch"
        )

        if not db_url:
            raise ValueError("TEST_DATABASE_URL not set and no default available")

        print(f"[{datetime.utcnow().isoformat()}] Initializing test database...")
        print(f"  Database URL: {db_url}")

        if dry_run:
            print("  [DRY RUN] Would execute:")
            print("    1. DROP all existing tables")
            print("    2. CREATE fresh schema from SQLModel definitions")
            if seed:
                print("    3. INSERT minimal seed data (2-3 test paddles)")
            return {"status": "dry-run", "message": "No changes applied"}

        # Create engine and drop/recreate tables
        engine = create_engine(db_url, echo=False)

        print("  Dropping existing tables...")
        SQLModel.metadata.drop_all(engine)

        print("  Creating fresh schema...")
        SQLModel.metadata.create_all(engine)

        # Optional: seed minimal test data
        if seed:
            print("  Seeding test data...")
            with Session(engine) as session:
                # Add 1-2 minimal paddle records for smoke testing
                test_paddle = PaddleMaster(
                    brand="TestPaddle",
                    model="Test-01",
                    specs_confidence=1.0,
                    source="test_db_init.py",
                )
                session.add(test_paddle)
                session.commit()
                print(f"    Added 1 test paddle record")

        print(f"[{datetime.utcnow().isoformat()}] Database initialization complete")
        return {"status": "success", "message": "Test database reset"}

    except ImportError as e:
        return {
            "status": "error",
            "message": f"Could not import SQLModel or models: {str(e)}",
            "note": "Ensure app/models.py exists and SQLModel is installed"
        }
    except Exception as e:
        return {
            "status": "error",
            "message": f"Database initialization failed: {str(e)}",
            "db_url": os.environ.get("TEST_DATABASE_URL", "not set"),
        }


def main():
    parser = argparse.ArgumentParser(
        description="Reset test database to fresh state for audit testing"
    )
    parser.add_argument(
        "--seed",
        action="store_true",
        help="Seed minimal test data after reset"
    )
    parser.add_argument(
        "--dry-run",
        action="store_true",
        help="Show what would happen without making changes"
    )

    args = parser.parse_args()

    result = init_test_db(seed=args.seed, dry_run=args.dry_run)

    if result["status"] in ("success", "dry-run"):
        print(f"✓ {result['message']}")
        return 0
    else:
        print(f"✗ {result['message']}")
        if "note" in result:
            print(f"  {result['note']}")
        return 1


if __name__ == "__main__":
    sys.exit(main())
