**Analysis algorithm:**

For each entry in execution_log.json:
1. Extract: script, status, error_category, error_reason, timestamp
2. If status="PASS":
   - remediation_type = "NONE"
   - priority = 0 (no action needed)
   - set last_successful_run = timestamp
3. If status="FAIL" or "TIMEOUT":
   - Map error_category to remediation_type:
     * NETWORK, TIMEOUT → "ADD_RETRY_LOGIC" (priority 2)
     * PARSING, PLAYWRIGHT → "FIX_SELECTORS_OR_PARSING" (priority 2)
     * API → "FIX_AUTHENTICATION_OR_ENDPOINT" (priority 1 — blocks all runs)
     * SCHEMA → "VALIDATE_DATABASE_SCHEMA" (priority 1 — data corruption risk)
     * DEPENDENCY → "INSTALL_MISSING_DEPENDENCY" (priority 1 — can't run at all)
     * FILE → "CHECK_INPUT_FILES_AND_PATHS" (priority 2)
     * UNKNOWN → "INVESTIGATE_AND_DEBUG" (priority 3 — needs manual review)
   - Extract first N lines of stderr (up to 200 chars) as root_cause_detail
   - Set last_successful_run = null or earliest known run if this is first execution
4. For transient failures (NETWORK, TIMEOUT), note in remediation that retry_strategy may help

```python
import json
from pathlib import Path
from datetime import datetime

def analyze_execution_log(log_path=".audit/execution_log.json"):
    with open(log_path) as f:
        log = json.load(f)

    analysis = {}

    for entry in log:
        script = entry["script"]

        if entry["status"] == "PASS":
            remediation_type = "NONE"
            priority = 0
            last_successful = entry["timestamp"]
        else:
            category = entry.get("error_category", "UNKNOWN")

            # Map category to remediation type
            remediation_map = {
                "NETWORK": "ADD_RETRY_LOGIC",
                "TIMEOUT": "ADD_RETRY_LOGIC",
                "PARSING": "FIX_SELECTORS_OR_PARSING",
                "PLAYWRIGHT": "FIX_SELECTORS_OR_PARSING",
                "API": "FIX_AUTHENTICATION_OR_ENDPOINT",
                "SCHEMA": "VALIDATE_DATABASE_SCHEMA",
                "DEPENDENCY": "INSTALL_MISSING_DEPENDENCY",
                "FILE": "CHECK_INPUT_FILES_AND_PATHS",
                "UNKNOWN": "INVESTIGATE_AND_DEBUG",
            }

            remediation_type = remediation_map.get(category, "INVESTIGATE_AND_DEBUG")

            # Priority mapping
            priority_map = {
                "INSTALL_MISSING_DEPENDENCY": 1,
                "FIX_AUTHENTICATION_OR_ENDPOINT": 1,
                "VALIDATE_DATABASE_SCHEMA": 1,
                "ADD_RETRY_LOGIC": 2,
                "FIX_SELECTORS_OR_PARSING": 2,
                "CHECK_INPUT_FILES_AND_PATHS": 2,
                "INVESTIGATE_AND_DEBUG": 3,
            }

            priority = priority_map.get(remediation_type, 3)
            last_successful = None  # Will need to search logs for this

        analysis[script] = {
            "status": entry["status"],
            "exit_code": entry.get("exit_code", "N/A"),
            "error_category": entry.get("error_category", "N/A"),
            "root_cause_detail": entry.get("stderr", "")[:200],  # First 200 chars
            "remediation_type": remediation_type,
            "priority": priority,
            "is_transient": entry.get("is_transient", False),
            "last_successful_run": last_successful,
            "timestamp": entry["timestamp"],
        }

    return analysis

if __name__ == "__main__":
    analysis = analyze_execution_log()
    with open(".audit/root_cause_analysis.json", "w") as f:
        json.dump(analysis, f, indent=2)
    print(f"Root cause analysis saved for {len(analysis)} scrapers")
```

**Important notes:**
- Extract first 200 chars of stderr to avoid bloat
- last_successful_run may be null for first-time audit
- priority values: 1=blocking (must fix immediately), 2=important (Phase 2), 3=investigate (possibly deferred)
- is_transient helps identify scrapers that may succeed on retry without code changes

```python
import json
from pathlib import Path
from datetime import datetime

def generate_detailed_report():
    with open(".audit/root_cause_analysis.json") as f:
        analysis = json.load(f)

    with open(".audit/execution_log.json") as f:
        log = json.load(f)

    # Categorize by status
    passing = {k: v for k, v in analysis.items() if v["status"] == "PASS"}
    failing = {k: v for k, v in analysis.items() if v["status"] != "PASS"}

    # Count by remediation type
    remediation_counts = {}
    for v in analysis.values():
        rm = v["remediation_type"]
        remediation_counts[rm] = remediation_counts.get(rm, 0) + 1

    # Generate report
    lines = [
        "# Scraper Health Audit Report",
        f"Generated: {datetime.utcnow().isoformat()}",
        "",
        "## Executive Summary",
        f"- Total scrapers: {len(analysis)}",
        f"- Passing: {len(passing)} ✓",
        f"- Failing: {len(failing)} ✗",
        "",
        "## Remediation Breakdown",
    ]

    for rem_type, count in sorted(remediation_counts.items()):
        lines.append(f"- {rem_type}: {count}")

    lines.extend([
        "",
        "## Detailed Per-Scraper Analysis",
        ""
    ])

    # Group by scraper category (infer from filename)
    shopify_scrapers = [k for k in analysis.keys() if "scrape_" in k and "justpaddles" not in k and "brain" not in k]
    playwright_scrapers = [k for k in analysis.keys() if "justpaddles" in k or "brain" in k]
    csv_scrapers = [k for k in analysis.keys() if "ingest_" in k or "fetch_" in k or "enrich_" in k]

    for category_name, scrapers in [
        ("Shopify API Scrapers", shopify_scrapers),
        ("Playwright Browser Scrapers", playwright_scrapers),
        ("CSV & Specialized Scrapers", csv_scrapers),
    ]:
        if not scrapers:
            continue

        lines.extend([f"### {category_name}", ""])

        for scraper in scrapers:
            data = analysis[scraper]
            status_icon = "✓" if data["status"] == "PASS" else "✗"

            lines.extend([
                f"#### {scraper}",
                f"- **Status:** {data['status']} {status_icon}",
                f"- **Exit Code:** {data['exit_code']}",
                f"- **Error Category:** {data['error_category']}",
            ])

            if data["remediation_type"] != "NONE":
                lines.extend([
                    f"- **Remediation Type:** {data['remediation_type']}",
                    f"- **Priority:** {data['priority']}",
                    f"- **Root Cause:** {data['root_cause_detail'][:100]}..." if data['root_cause_detail'] else "- **Root Cause:** Unknown",
                ])
            else:
                lines.append("- **Remediation:** None — scraper is healthy")

            lines.append("")

    # Production readiness
    lines.extend([
        "## Production Readiness",
        "",
        "### Healthy (PASS)",
    ])

    for scraper in passing:
        lines.append(f"- {scraper}")

    if failing:
        lines.extend(["", "### Requires Fixes"])
        for scraper, data in sorted(failing.items(), key=lambda x: x[1]["priority"]):
            lines.append(f"- {scraper} ({data['remediation_type']}, P{data['priority']})")

    return "\n".join(lines)

if __name__ == "__main__":
    report = generate_detailed_report()
    with open(".audit/detailed_audit_report.md", "w") as f:
        f.write(report)
    print(f"Detailed audit report generated: {len(report)} characters")
```

```python
import json
from pathlib import Path
from datetime import datetime
from collections import defaultdict

def generate_health_summary():
    with open(".audit/root_cause_analysis.json") as f:
        analysis = json.load(f)

    # Metrics
    passing = [s for s, d in analysis.items() if d["status"] == "PASS"]
    failing = [s for s, d in analysis.items() if d["status"] != "PASS"]

    # Failure breakdown
    failure_breakdown = defaultdict(int)
    for data in analysis.values():
        if data["status"] != "PASS":
            cat = data.get("error_category", "UNKNOWN")
            failure_breakdown[cat] += 1

    # Organize by priority
    priority_1 = {}
    priority_2 = {}
    priority_3 = {}

    for scraper, data in analysis.items():
        if data["status"] == "PASS":
            continue

        priority = data["priority"]
        item = {
            "scraper": scraper,
            "issue": data["remediation_type"],
            "error_category": data["error_category"],
            "last_run": data["timestamp"],
        }

        if priority == 1:
            priority_1[scraper] = item
        elif priority == 2:
            priority_2[scraper] = item
        else:
            priority_3[scraper] = item

    # Production safe
    production_safe = [
        {
            "scraper": s,
            "status": analysis[s]["status"],
            "last_successful_run": analysis[s]["last_successful_run"],
            "error_category": "SUCCESS"
        }
        for s in passing
    ]

    # Transient failures
    transient = [
        {
            "scraper": s,
            "error_category": data["error_category"],
            "is_transient": data["is_transient"],
            "recommendation": "Retry without code changes; may self-heal"
        }
        for s, data in analysis.items()
        if data.get("is_transient") and data["status"] != "PASS"
    ]

    # Recommendations
    recommendations = []
    if priority_1:
        recommendations.append(f"Priority 1: Fix {len(priority_1)} blocking issues")
    if priority_2:
        recommendations.append(f"Priority 2: Address {len(priority_2)} important fixes (retry logic, selectors)")
    if transient:
        recommendations.append(f"Monitor: {len(transient)} transient failures that may auto-recover")

    summary = {
        "audit_timestamp": datetime.utcnow().isoformat(),
        "total_scrapers": len(analysis),
        "health_metrics": {
            "passing": len(passing),
            "failing": len(failing),
            "unknown": 0,
        },
        "failure_breakdown": dict(failure_breakdown),
        "remediation_priority": {
            "PRIORITY_1_BLOCKING": list(priority_1.values()),
            "PRIORITY_2_IMPORTANT": list(priority_2.values()),
            "PRIORITY_3_INVESTIGATE": list(priority_3.values()),
        },
        "production_safe": production_safe,
        "transient_failures": transient,
        "last_audit_run": datetime.utcnow().isoformat(),
        "recommendations": recommendations,
    }

    return summary

if __name__ == "__main__":
    summary = generate_health_summary()
    with open(".audit/scraper_health_summary.json", "w") as f:
        json.dump(summary, f, indent=2)
    print(f"Health summary saved")
```
