**Error categories to implement (from RESEARCH.md Pitfalls):**
1. NETWORK — "HTTP 429", "Connection timeout", "Name or service not known", "Failed to resolve"
2. PARSING — "Selector", "JSON decode", "KeyError", "IndexError", "attribute error" (case-insensitive)
3. API — "Unauthorized", "403", "404", "Authentication failed", "Invalid API key"
4. SCHEMA — "NOT NULL", "Unique constraint", "Column does not exist", "Database error"
5. TIMEOUT — "TimeoutError", "timed out", "exceeded deadline", "504"
6. PLAYWRIGHT — "Browser not found", "OSError: browser", "chromium"
7. DEPENDENCY — "ModuleNotFoundError", "ImportError", "No module named"
8. FILE — "No such file", "FileNotFoundError", "cannot open"
9. UNKNOWN — catch-all if no pattern matches

**Logic:**
- Check stderr line-by-line for patterns (compile regex once, reuse)
- Set `is_transient=True` for NETWORK, TIMEOUT, PLAYWRIGHT (can retry)
- Set `is_transient=False` for PARSING, API, SCHEMA, DEPENDENCY, FILE, UNKNOWN
- If exit_code=0, return `{category: "SUCCESS", reason: "Exit code 0", is_transient: False}`
- If exit_code!=0 but stderr empty, return `{category: "UNKNOWN", reason: f"Exit code {exit_code}, no stderr output", is_transient: False}`

**Output format example:**
```python
{
  "category": "NETWORK",
  "reason": "HTTP 429 Too Many Requests",
  "is_transient": True,
  "matched_pattern": "HTTP 429"
}
```

Do NOT implement retry logic here—only classification.

```python
import subprocess
import json
from pathlib import Path
from datetime import datetime
from .audit import error_categorization

SCRAPERS = [
    "scrape_joola.py",
    "scrape_shark.py",
    # ... all 11, insert actual names after verification
]

TIMEOUT_SECONDS = 300  # 5 minutes per scraper max
LOG_FILE = Path(".audit/execution_log.json")
STATUS_FILE = Path(".audit/status_matrix.md")

def run_scraper(script_name: str) -> dict:
    """Execute single scraper, capture all metadata."""
    try:
        result = subprocess.run(
            ["docker", "compose", "exec", "-T", "backend_v3", "python", f"scripts/{script_name}"],
            capture_output=True,
            text=True,
            timeout=TIMEOUT_SECONDS,
            cwd="/path/to/project"  # Set to sliceinsights root
        )

        error_cat = error_categorization.categorize_error(result.stderr, result.returncode)

        return {
            "script": script_name,
            "exit_code": result.returncode,
            "stdout": result.stdout[:5000],  # Truncate to 5KB to prevent log bloat
            "stderr": result.stderr[:5000],
            "timestamp": datetime.utcnow().isoformat(),
            "status": "PASS" if result.returncode == 0 else "FAIL",
            "error_category": error_cat["category"],
            "error_reason": error_cat["reason"],
            "is_transient": error_cat["is_transient"],
        }
    except subprocess.TimeoutExpired:
        return {
            "script": script_name,
            "status": "TIMEOUT",
            "error_category": "TIMEOUT",
            "error_reason": f"Scraper exceeded {TIMEOUT_SECONDS}s timeout",
            "is_transient": True,
            "timestamp": datetime.utcnow().isoformat(),
        }
    except Exception as e:
        return {
            "script": script_name,
            "status": "ERROR",
            "error_category": "UNKNOWN",
            "error_reason": f"Harness error: {str(e)}",
            "is_transient": False,
            "timestamp": datetime.utcnow().isoformat(),
        }

def run_all_scrapers():
    """Execute all scrapers and save to execution_log.json."""
    LOG_FILE.parent.mkdir(parents=True, exist_ok=True)
    execution_log = []

    print(f"Starting audit of {len(SCRAPERS)} scrapers at {datetime.utcnow().isoformat()}")
    for i, scraper in enumerate(SCRAPERS, 1):
        print(f"  [{i}/{len(SCRAPERS)}] Running {scraper}...")
        result = run_scraper(scraper)
        execution_log.append(result)
        print(f"    → {result['status']}: {result.get('error_category', 'N/A')}")

    # Save execution log
    with open(LOG_FILE, "w") as f:
        json.dump(execution_log, f, indent=2)
    print(f"\nExecution log saved to {LOG_FILE}")

    return execution_log

def generate_status_matrix(execution_log: list) -> str:
    """Generate markdown status matrix from execution log."""
    lines = [
        "# Scraper Health Status Matrix",
        "",
        "| Scraper | Status | Error Category | Transient? | Last Run |",
        "|---------|--------|----------------|-----------|----------|",
    ]

    for entry in execution_log:
        script = entry["script"]
        status = entry["status"]
        category = entry.get("error_category", "N/A")
        transient = "Yes" if entry.get("is_transient") else "No"
        timestamp = entry.get("timestamp", "N/A")[:10]  # Date only

        lines.append(f"| {script} | {status} | {category} | {transient} | {timestamp} |")

    return "\n".join(lines)

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--output", default=".audit/status_matrix.md", help="Output file for status matrix")
    args = parser.parse_args()

    log = run_all_scrapers()
    matrix = generate_status_matrix(log)

    with open(args.output, "w") as f:
        f.write(matrix)
    print(f"Status matrix saved to {args.output}")
```

**Important details:**
- Use `-T` flag in docker compose exec (non-interactive mode, prevents tty allocation issues)
- Set cwd to project root to ensure relative paths work
- Truncate stdout/stderr to 5KB each to keep JSON manageable
- Initialize `.audit/` directory automatically
- Log timestamp in ISO format for sorting/analysis
- Call error_categorization.categorize_error() for every execution
