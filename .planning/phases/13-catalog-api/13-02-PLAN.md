**CRITICAL: Mock session strategy for catalog endpoints.**

The catalog `list_catalog_paddles` endpoint makes 2 `session.exec()` calls:
- First call: data query returning PaddleMaster objects (with loaded brand, market_offers, and market_offers[].store relationships)
- Second call: count query returning an integer

So `mock.exec.side_effect` must be a list of 2 mock results.

For the data query result: The endpoint iterates over `result.all()` which returns PaddleMaster objects. Each PaddleMaster must have:
- `.brand.name` (string)
- `.market_offers` (list of MarketOffer, each with `.store.name`, `.store.slug`, `.price_brl`, `.url`, `.is_active`)
- `.core_thickness_mm`, `.face_material`, `.model_name`, `.id`

```python
@pytest.mark.asyncio
async def test_catalog_paddles_returns_list():
    """GET /catalog/paddles returns paddles with store URLs — CAT-01, CAT-06."""
    from tests.conftest import MockPaddle, MockStore, MockMarketOffer

    store = MockStore(id=1, name="ProPadel", slug="propadel")
    paddle1 = MockPaddle(name="Hyperion CFS", brand_name="Joola")
    paddle1.market_offers = [
        MockMarketOffer(paddle_id=paddle1.id, price=899.90, store=store)
    ]
    paddle2 = MockPaddle(name="Vanguard", brand_name="Selkirk")
    paddle2.market_offers = [
        MockMarketOffer(paddle_id=paddle2.id, price=1299.00, store=store)
    ]

    async def mock_session_gen():
        mock = AsyncMock()
        data_result = MagicMock()
        data_result.all.return_value = [paddle1, paddle2]
        count_result = MagicMock()
        count_result.first.return_value = 2
        mock.exec.side_effect = [data_result, count_result]
        yield mock

    app.dependency_overrides[get_session] = mock_session_gen
    transport = ASGITransport(app=app)
    async with AsyncClient(transport=transport, base_url="http://test") as client:
        response = await client.get("/api/v1/catalog/paddles")
    app.dependency_overrides.clear()

    assert response.status_code == 200
    body = response.json()
    assert "data" in body
    assert "total" in body
    assert body["total"] == 2
    assert len(body["data"]) == 2
    p1 = body["data"][0]
    assert "brand" in p1
    assert "model_name" in p1
    assert "specs" in p1
    assert "market_offers" in p1
    assert len(p1["market_offers"]) >= 1
    assert "store_url" in p1["market_offers"][0]
```
