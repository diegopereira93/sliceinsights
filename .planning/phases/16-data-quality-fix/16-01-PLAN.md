### 2. Deletar 5 paddles com fotos falsas

```bash
docker exec picklematch_api_prod python -c "
from sqlmodel import Session, select
from app.db.database import sync_engine
from app.models.paddle import PaddleMaster
from app.models.market_offer import MarketOffer
import uuid

BAD_UUIDS = [
    '88c7349f-...',
    'b6e3b220-...',
    '294b1891-...',
    'e1629234-...',
    'cbf380dd-...',
]

with Session(sync_engine) as s:
    for uid_str in BAD_UUIDS:
        uid = uuid.UUID(uid_str)
        offers = s.exec(select(MarketOffer).where(MarketOffer.paddle_id == uid)).all()
        for o in offers:
            s.delete(o)
        paddle = s.get(PaddleMaster, uid)
        if paddle:
            print(f'Deleted: {paddle.model_name}')
            s.delete(paddle)
    s.commit()
"
```

### 3. Corrigir brand names

```bash
docker exec picklematch_api_prod python -c "
from sqlmodel import Session, select
from app.db.database import sync_engine
from app.models.brand import Brand
from app.models.paddle import PaddleMaster

RENAMES = {
    '3Rdshot': '3RD Shot',
    'Slk': 'SLK',
    'Com': None,       # marcar para delete
    'Cs': None,        # marcar para delete
    'Pulse': 'Pulse',
    'Eagle': 'Eagle',
    'Falcon': 'Falcon',
}

with Session(sync_engine) as s:
    # Renames
    for old, new in RENAMES.items():
        if new is None:
            continue
        brand = s.exec(select(Brand).where(Brand.name == old)).first()
        if brand:
            print(f'Rename: {old} -> {new}')
            brand.name = new
            s.add(brand)
    s.commit()
"
```

### 4. Lidar com brands sem paddles (Com, Cs)

```bash
docker exec picklematch_api_prod python -c "
from sqlmodel import Session, select
from app.db.database import sync_engine
from app.models.brand import Brand
from app.models.paddle import PaddleMaster

# Verificar se Com e Cs têm paddles
with Session(sync_engine) as s:
    for name in ['Com', 'Cs']:
        brand = s.exec(select(Brand).where(Brand.name == name)).first()
        if brand:
            paddles = s.exec(select(PaddleMaster).where(PaddleMaster.brand_id == brand.id)).all()
            print(f'{name}: {len(paddles)} paddles')
            for p in paddles:
                print(f'  - {p.model_name}')
"
```
