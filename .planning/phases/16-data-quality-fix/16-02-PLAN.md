**Option A — If keeping Com (real brand found):**
Rename the brand and reassign paddle:
```bash
docker exec picklematch_api_prod python -c "
from sqlmodel import Session, select
from app.db.database import sync_engine
from app.models.brand import Brand
import uuid

REAL_BRAND_NAME = '{COM_REAL_BRAND}'  # Fill in from Task 2

with Session(sync_engine) as s:
    old_brand = s.exec(select(Brand).where(Brand.name == 'Com')).first()
    if old_brand:
        # Check if real brand already exists
        real_brand = s.exec(select(Brand).where(Brand.name == REAL_BRAND_NAME)).first()
        if not real_brand:
            real_brand = Brand(name=REAL_BRAND_NAME)
            s.add(real_brand)
            s.flush()
        # Reassign paddles
        from app.models.paddle import PaddleMaster
        paddles = s.exec(select(PaddleMaster).where(PaddleMaster.brand_id == old_brand.id)).all()
        for p in paddles:
            p.brand_id = real_brand.id
            s.add(p)
        s.delete(old_brand)
        s.commit()
        print(f'Merged Com -> {REAL_BRAND_NAME} ({len(paddles)} paddles)')
    else:
        print('Com brand not found')
"
```
